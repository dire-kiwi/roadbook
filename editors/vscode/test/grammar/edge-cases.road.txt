extern type Clock
enum Level { Low, High }
type Settings { label: string?, ids: int[]?, peer: ref Worker, ratio: float }

task fetch(
  url: string,
  retries: int,
) -> json ! Timeout | Refused

machine Worker(settings: Settings, peer: ref Worker) {
  var ratio: float = 0.5
  var text: string = "tab\t quote\" backslash\\ newline\n bad\q"
  emits Ready { level: Level }

  state Paused {
    initial Idle
    /// Back to where `Paused` was.
    history Last
    deep history All
    state Idle { on * -> Paused.Last }
  }

  parallel state Running {
    state A {
      on * -> Running.B
      always if not (ratio >= 1.25 and ratio != -2.0) or true -> B set ratio = ratio * 2.0
    }
    state B {
      entry do audit("b"), log(ratio)
      after 250ms -> A
      after 2h { if ratio < 1.0 -> A else -> B }
      on peer.Ready if event.level == Level.High -> A emit Ready { level: Level.Low }
      invoke Child(settings) as child {
        done -> A send peer.Ping {}
        fail * -> B
      }
    }
    state C {
      invoke fetch(retry) retry 3 on Busy | Gone after backoff(attempt) {
        done -> A
        fail * -> B
      }
    }
  }
  final state Broken { fail error }
}

machine Stock() {
  /// A call this machine answers.
  call reserve(orderId: string, items: Item[]) -> Reservation ! OutOfStock | Closed
  call count() -> int
  var pending: call reserve

  on count reply call done held
  state Ready {
    on reserve -> Busy  set pending = call, orderId = event.orderId
    on count if held > 0 reply call fail OutOfStock { reason: "busy" }
  }
  state Busy {
    invoke reserveStock(orderId) {
      done            -> Ready  reply pending done Reservation { orderId: orderId }
      fail OutOfStock -> Ready  reply pending fail error
    }
  }
  state Asking {
    invoke peer.reserve(id, items) {
      done            -> Ready  reply pending done
      fail Unanswered -> Ready
      fail *          -> Busy
    }
  }
}

machine Hub(children: int) {
  children sessions: Session by string
  initial A
  exit send sessions.*.Leave
  state A {
    on Join             spawn sessions[event.id] = Session(event.name)  send sessions[event.id].Notify { key: key }
    on Kick             stop sessions[event.id]  do stop(), spawn(children)
    on sessions.done    do log(key, result, event.key)
    on sessions.fail *  -> A
    on sessions.fail Lost -> A
  }
}

machine Checked(check: int) {
  var never: int = check
  initial A
  state A {
    initial B
    state B {}
  }
  /// Only after A.
  check A.B only after A
  check reachable A.B // a comment
  check never A and A.B
  check can finish
  on Go set check = never
}
machine Watch(server: string, folder: string) {
  initial Watching
  state Watching {
    invoke heartbeat(server) as beat { fail Lost -> Offline }
    invoke syncFiles(folder) as sync {
      done   -> Synced
      fail * -> Offline
    }
    invoke ticks(1s) as clock
  }
  final state Synced {}
  final state Offline {}
}

// One transition for several events.
machine Several() {
  on Reset | Drop -> A
  state A {
    on Retry|Reconnect if attempts < 3 -> A  set reason = event.reason
  }
}

// History states with default targets.
machine Defaults() {
  state A {
    initial B
    history Last -> B
    deep history All->B.C, D
    state B {}
  }
}

// Choice states, and `choice` as a name.
machine Choices() {
  var choice: int = 0
  choice Pick {
    if choice > 1 -> A  do log()
    else          -> B
  }
}

// Whether a state is active: `in(…)` in a guard, a target's arguments and an effect.
machine Ins() {
  state S {
    on Go if not in(Main.Volume.Muted) -> Paused(in(Main.Volume.Loud))  do note(in(Main))
  }
}
