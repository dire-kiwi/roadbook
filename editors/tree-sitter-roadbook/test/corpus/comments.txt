// A line comment.
//// Four slashes are a line comment too.
///
/// Documents the type.
type T { a: int } // after code

/// Documents the machine.
machine M() {
  /// Documents the variable.
  var x: int = 0 // trailing
  initial A // c
  /// Documents the state.
  state A {
    // inside
    on X -> A
  }
}
