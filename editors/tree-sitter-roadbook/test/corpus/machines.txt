  initial Running

  parallel state Running {
    state Left {
      initial A
      state A { on Go -> B }
      final state B {}
    }
    state Right {
      initial C
      state C {
        initial D
        state D {}
      }
    }
    done -> Finished
  }

  final state Finished { output 1 }
  final state Failed   { fail Broken }
}
