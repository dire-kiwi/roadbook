  initial Waiting
  state Waiting {
    invoke ticks(1s) as clock
    on clock.Tick set seconds = seconds + 1
  }
  state Uploading {
    invoke upload(file) as up retry 2 after 1s
    after 5s -> Waiting
  }
}

  initial Open
  state Open {
    on Add spawn uploads[event.file.name] = upload(event.file)
    on uploads.Progress do show(key, event.percent)
