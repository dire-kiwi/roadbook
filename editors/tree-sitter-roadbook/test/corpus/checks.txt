machine PaymentWebhook(check: int) {
  var never: int = check
  initial Applying
  state Applying {
    initial RecordingRefund
    state RecordingRefund {}
  }
  /// A refund only after the duplicate check.
  check Applying.RecordingRefund only after CheckingDuplicate
  check reachable Applying.RecordingRefund
  check never BadSignature
  check never Applying.A and Applying.B and C
  check can finish
  after 5s -> Applying
}

strict machine Store() {
  call reserve(orderId: string) -> int ! OutOfStock
  check reachable Open
  initial Open
  on reserve reply call done
  check never Open and Closed
  on Ping ignore
  check can finish
}
