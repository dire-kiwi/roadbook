Internal events, raise and cancel

machine M() {
  event Go
  /// Sent by the machine to itself.
  internal event Remind { attempt: int }
  internal event Ping
  on Go raise Ping  raise Remind { attempt: 1 } after 24h as reminder
  on Ping cancel reminder  set n = n + 1
  state A {
    entry raise Remind { attempt: 2 } after delay(n)
    exit cancel reminder  do log()
    on Remind if n > 1 -> A  raise Ping
    always {
      if n > 2 raise Ping after 1s
      else cancel reminder
    }
  }
}

    after 5s -> A
    on Stop raise Ping { n: 1 }
    after 5s do f()
    on Key raise Check after 500ms as debounce  set n = 0
    after 1s raise Ping after 2s
  }
}
