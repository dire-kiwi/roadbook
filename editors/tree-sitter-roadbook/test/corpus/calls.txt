================================================================================
Invoking a call on a peer
================================================================================

machine Order(id: string, inventory: ref Inventory) {
  initial Reserving
  state Reserving {
    invoke inventory.reserve(id, items) {
      done            -> Reserved  set reservation = result
      fail OutOfStock -> Failed    set problem = error
      fail Unanswered -> Failed
    }
    invoke inventory.count() {
      fail * -> Gave
    }
  }
}

================================================================================
Reply effects: done with and without a value, fail, and what follows them
================================================================================

machine Inventory() {
  on count reply call done
  call count() -> int
  on ping reply call done
  event Close
  on stop reply pending done  reply pending fail OutOfStock { reason: "second" }
  state Reserving {
    invoke reserveStock(orderId) {
      done -> Ready  set held = held + 1  reply pending done Reservation { orderId: orderId, count: result }
      fail OutOfStock -> Ready  reply pending fail error
    }
    on reserve reply call done
    on Close   -> Closed  reply pending done  emit Closed
  }
}
