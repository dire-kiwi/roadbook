Spawn, stop and send by key or to all
================================================================================

machine Hub() {
  initial A
  exit send sessions.*.Leave
  state A {
    entry spawn jobs[Lane.Fast] = Job(Lane.Fast)
    on Connected     spawn sessions[event.id] = Session(event.id, "hi")  send sessions[event.id].Notify { text: "hi" }
    on Disconnected  stop sessions[event.id]
    on Broadcast     send sessions.*.Notify { text: event.text }
  }
}

  initial A
  state A {
    on sessions.Typing                      do note(key, event.text)
    on sessions.done                        set open = open - 1  do note(key, result)
    on sessions.fail Lost if error.reason == "idle"  do log(key)
    on sessions.fail *                      -> A
  }
